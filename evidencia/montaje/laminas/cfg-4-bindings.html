Tres bindings por flujo

config/RabbitTopologyConfig.java — ejemplo del flujo de cocina
 1@Bean  // clave exacta: un ticket tiene un unico destinatario
 2Binding bindKitchenDirect() {
 3    return BindingBuilder.bind(colaKitchen())
 4            .to(cmdDirectExchange()).with(routingKey(KITCHEN));
 5}
 6 
 7@Bean  // patron con comodines: admite variantes futuras
 8Binding bindKitchenTopic() {
 9    return BindingBuilder.bind(colaKitchen())
10            .to(cmdTopicExchange()).with(patronTopic(KITCHEN));
11}
12 
13@Bean  // la cola muerta cuelga del dead-letter exchange
14Binding bindKitchenDlq() {
15    return BindingBuilder.bind(colaKitchenDlq())
16            .to(cmdDeadLetterExchange()).with(routingKey(KITCHEN));
17}
